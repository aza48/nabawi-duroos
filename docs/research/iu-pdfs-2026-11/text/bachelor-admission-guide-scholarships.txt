                                               ‫المنح الدراسية في الجامعة الإسلامية‬

   ‫تمثـل المنـح الدراسـية فـي الجامعـة الإسلاميـة بالمدينـة المنـورة إحـدى المبـادرات التعليميـة‬
   ‫التـي تهـدف إلـى توفيـر فـرص أكاديميـة متميـزة للـطلاب المؤهليـن مـن مختلـف دول العالـم،‬
   ‫ضمــن بيئــة جامعيــة متكاملــة تدعــم التحصيــل العلمــي، وتنميــة المهــارات، وتعــزز التبــادل‬

                                ‫الثقافــي والمعرفــي بيــن الــطلاب مــن مختلــف الخلفيــات والثقافــات.‬

   ‫ومــن خلال برامــج المنــح الدراســية، تســعى الجامعــة إلــى استقطــاب الــطلاب المتميزيــن‬
   ‫أكاديمًيًــا، وتمكينهــم مــن الالتحــاق بالتخصصــات والبرامــج الأكاديميــة المختلفــة، والاســتفادة‬
   ‫مــن الخدمــات التعليميــة والبحثيــة والطلابيــة التــي توفرهــا الجامعــة وفــق أعلــى المعاييــر‬

                                                                                      ‫الأكاديميــة والتنظيميــة.‬

   ‫ومــن خلال هــذه المنــح، تواصــل الجامعــة الإسلاميــة رســالتها التعليميــة العالميــة الممتــدة‬
   ‫منــذ تأسيســها، فــي إعــداد خريجيــن مؤهليــن علمًيًــا ومعرفًيًــا، قادريــن علــى الإســهام الإيجابــي‬
   ‫فـي مجتمعاتهـم، ونقـل المعرفـة، وتعزيـز قيـم الوسـطية والتعـاون والتواصـل الحضـاري بيـن‬

                                        ‫أوًالً: المنح الدراسية الكاملة )المنح الخارجية(‬

   ‫هـــي الــمــقـــعـــد الـــدراســــي الـــذي يـــحـــصــــل عـــلـــيـــه الــطـــالـــب الـمـــقــيـــم خـــارج الـــمــمــلــكــة مــن غير‬
           ‫السـعوديين؛ للدراسـة فـي مؤسسـات التعليـم العالـي فـي المملكـة العربيـة السـعودية.‬

   ‫وهـــو مـــقـــعـــد مـــجــــانـــي للــطــــالــــب فــــي مـــــرحــــلــــة الـــبــــكــــالــــوريــــوس )نــــــوع الــــــدراســــة: انـــتـــظـــام(‬
   ‫تــــمــــنـــحـــه حــــكـــومـــة الــــــمــــمــــلــــكـــة الــــــعـــــربـــــيــــة الــــســــعــــوديــــة للـــــطــــــلاب الـــدولــيــيــن، مــــع تــــقـــديــــم‬

           ‫عـــــدد مــــــن الــــمــــزايــــــا الـــمـــالــيـــة، وتـــــوفــــيـــر الـــخـــدمـــات الـــمــــلائــــمـــة للبــــيــــئـــة الــــجــــامـــعــيـــــة.‬
